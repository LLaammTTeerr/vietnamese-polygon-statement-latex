\begin{problem}[
  input = stdin, output = stdout,
  time = 1, memory = 256, points = 100,
  author = Lâm, origin = {VOI 2024},
]{Dãy con tăng dài nhất}

Cho dãy số nguyên $a_1, \dots, a_n$.

\InputFile
Dòng đầu chứa số nguyên $n$.

\OutputFile
In ra một số nguyên duy nhất.

\Constraints
$1 \le n \le 10^5$.

\begin{subtasks}
  \subtask{30}{$n \le 20$}
  \subtask{30}{$n \le 2000$}
  \subtask{40}{Không có ràng buộc thêm}
\end{subtasks}

\Examples

\end{example}

\Explanation
Đáp án của ví dụ là $6$.
