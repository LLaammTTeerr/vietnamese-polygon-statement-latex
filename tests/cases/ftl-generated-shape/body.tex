\begin{problem}[
  input  = stdin,
  output = stdout,
  time   = 1,
  memory = 256,
]{Dãy con tăng dài nhất}

Cho dãy số nguyên $a_1, \dots, a_n$.

\InputFile
Dòng đầu chứa số nguyên $n$.

\OutputFile
In ra một số nguyên duy nhất.

\Examples
\begin{example}
\exmpfile{example.01}{example.01.a}%
\end{example}

\Note
Đáp án là $6$.
\end{problem}
